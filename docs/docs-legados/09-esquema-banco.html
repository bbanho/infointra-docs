09 — Esquema do Banco ​

Documento original

Conteúdo integral preservado de docs/09-esquema-banco-de-dados.md. Parte da wiki para manter a rastreabilidade com a documentação histórica do projeto.

Este documento contém o script SQL necessário para criar as tabelas no Supabase. Como a aplicação opera de forma Offline-First com Padrão Outbox, o Supabase atua como o backend em nuvem consolidado.

Instruções de Instalação ​

  1. Acesse o painel do Supabase do seu projeto.
  2. Vá para o SQL Editor.
  3. Copie o script abaixo e execute-o.
sql
-- Habilita extensão para UUIDs e RLS
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- 1. Tabela de Perfis de Usuários (Vendedores)
-- Vinculada à tabela auth.users do Supabase
CREATE TABLE IF NOT EXISTS public.vendedores (
    id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
    email TEXT UNIQUE NOT NULL,
    nome TEXT NOT NULL,
    cargo TEXT,
    codigo_vendedor TEXT UNIQUE NOT NULL,
    meta_mensal DECIMAL(15, 2) DEFAULT 0,
    comissao_percentual DECIMAL(5, 2) DEFAULT 0,
    permissao_desconto_max DECIMAL(5, 2) DEFAULT 0,
    avatar TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 2. Tabela de Clientes
CREATE TABLE IF NOT EXISTS public.clientes (
    id SERIAL PRIMARY KEY,
    tipo TEXT NOT NULL CHECK (tipo IN ('PF', 'PJ')),
    razao_social TEXT NOT NULL,
    nome_fantasia TEXT,
    cpf_cnpj TEXT UNIQUE NOT NULL,
    ie TEXT,
    email TEXT,
    whatsapp TEXT,
    telefone TEXT,
    cep TEXT,
    logradouro TEXT,
    numero TEXT,
    complemento TEXT,
    bairro TEXT,
    cidade TEXT,
    uf TEXT,
    limite_credito DECIMAL(15, 2) DEFAULT 0,
    credito_utilizado DECIMAL(15, 2) DEFAULT 0,
    status TEXT DEFAULT 'ATIVO' CHECK (status IN ('ATIVO', 'BLOQUEADO', 'INADIMPLENTE')),
    observacoes TEXT,
    data_cadastro TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 3. Tabela de Produtos (Catálogo)
CREATE TABLE IF NOT EXISTS public.produtos (
    id SERIAL PRIMARY KEY,
    sku TEXT UNIQUE NOT NULL,
    nome TEXT NOT NULL,
    categoria TEXT NOT NULL,
    is_service BOOLEAN DEFAULT false,
    preco DECIMAL(15, 2) NOT NULL,
    custo DECIMAL(15, 2) NOT NULL,
    estoque_atual DECIMAL(15, 2) DEFAULT 0,
    estoque_minimo DECIMAL(15, 2) DEFAULT 0,
    unidade TEXT NOT NULL,
    descricao TEXT,
    ativo BOOLEAN DEFAULT true,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 4. Tabela de Pedidos
CREATE TABLE IF NOT EXISTS public.pedidos (
    id SERIAL PRIMARY KEY,
    codigo_pedido TEXT UNIQUE NOT NULL,
    data_criacao TIMESTAMP WITH TIME ZONE NOT NULL,
    vendedor_id TEXT NOT NULL,
    vendedor_nome TEXT,
    cliente_id INTEGER REFERENCES public.clientes(id),
    cliente_nome TEXT,
    cliente_cpf_cnpj TEXT,
    cliente_cidade TEXT,
    cliente_uf TEXT,
    subtotal DECIMAL(15, 2) NOT NULL,
    desconto_geral_valor DECIMAL(15, 2) NOT NULL,
    total DECIMAL(15, 2) NOT NULL,
    forma_pagamento TEXT NOT NULL,
    parcelas INTEGER DEFAULT 1,
    condicao_pagamento TEXT,
    status_sync TEXT DEFAULT 'sincronizado',
    status_fiscal TEXT DEFAULT 'PENDENTE',
    chave_nfe TEXT,
    protocolo_nfe TEXT,
    cupom_fiscal_numero TEXT,
    observacoes_pedido TEXT,
    xml_gerado TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 5. Tabela de Itens do Pedido
CREATE TABLE IF NOT EXISTS public.pedido_itens (
    id SERIAL PRIMARY KEY,
    pedido_id INTEGER REFERENCES public.pedidos(id) ON DELETE CASCADE,
    produto_id INTEGER REFERENCES public.produtos(id),
    sku TEXT,
    nome TEXT,
    is_service BOOLEAN DEFAULT false,
    unidade TEXT,
    preco_unitario DECIMAL(15, 2) NOT NULL,
    quantidade DECIMAL(15, 2) NOT NULL,
    desconto_percentual DECIMAL(5, 2) DEFAULT 0,
    desconto_valor DECIMAL(15, 2) DEFAULT 0,
    subtotal DECIMAL(15, 2) NOT NULL,
    observacao TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- RLS (Row Level Security) - Regras de Segurança
ALTER TABLE public.vendedores ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.clientes ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.produtos ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.pedidos ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.pedido_itens ENABLE ROW LEVEL SECURITY;

-- Exemplo de Políticas Básicas
-- Criação de Políticas para permitir que vendedores autenticados leiam catálogo e clientes
CREATE POLICY "Permitir leitura de produtos para usuários autenticados" 
ON public.produtos FOR SELECT USING (auth.role() = 'authenticated');

CREATE POLICY "Permitir leitura de clientes para usuários autenticados" 
ON public.clientes FOR SELECT USING (auth.role() = 'authenticated');

-- Permitir leitura e gravação de pedidos pertencentes ao próprio vendedor (opcional, dependendo da regra de negócio)
CREATE POLICY "Vendedores podem gerenciar todos pedidos provisoriamente" 
ON public.pedidos FOR ALL USING (auth.role() = 'authenticated');

CREATE POLICY "Vendedores podem gerenciar itens de pedidos provisoriamente" 
ON public.pedido_itens FOR ALL USING (auth.role() = 'authenticated');

CREATE POLICY "Vendedores podem ler o próprio perfil" 
ON public.vendedores FOR SELECT USING (auth.uid() = id);
